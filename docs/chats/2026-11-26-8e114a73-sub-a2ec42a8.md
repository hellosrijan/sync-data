16	# "catch-up" is treated exactly like "salesforce" everywhere (AC-2.1.1).
17	SF_SOURCES = frozenset({"salesforce", "catch-up"})
18	MODELS = {"Customer": Customer, "Contact": Contact}
19	
20	
21	class UnknownRecord(LookupError):
22	    """A portal change or a Salesforce update names a record that does not exist. Raised rather
23	    than silently created (DEFAULTED, AC-2.1.2c); the transaction rolls back, so the ChangeEvent
24	    is not kept and a retry is not misreported as a duplicate."""
25	
26	
27	@dataclass(frozen=True)
28	class Change:
29	    source: str  # "salesforce" | "catch-up" | "portal"
30	    event_id: str
31	    object_type: str  # "Customer" | "Contact"
32	    record_ref: str | int  # salesforce_id (str) for salesforce/catch-up, pk (int) for portal
33	    occurred_at: datetime  # timezone-aware
34	    actor: str
35	    fields: dict[str, Any]
36	    operation: str  # "create" | "update"
37	
38	
39	@dataclass(frozen=True)
40	class Result:
41	    outcome: str  # "applied" | "no_change" | "duplicate" | "echo"
42	    applied_fields: list[str] = field(default_factory=list)
43	
44	
45	def apply(change: Change) -> Result:
46	    with transaction.atomic():
47	        try:
48	            # Savepoint, so a duplicate's IntegrityError doesn't poison the outer transaction.
49	            with transaction.atomic():
50	                ChangeEvent.objects.create(event_id=change.event_id, source=change.source)
51	        except IntegrityError:
52	            return Result(outcome="duplicate")
53	
54	        record, created = _lock_record(change)
55	
56	        if _is_echo(change):
57	            for name, value in change.fields.items():
58	                _write_history(change, record, name, value, FieldChange.Status.ECHO)
59	            return Result(outcome="echo")
60	
61	        applied_fields = []
62	        for name, value in _changed_fields(change, record).items():
63	            is_newer = _is_newer(change, record, name)
64	            status = FieldChange.Status.APPLIED if is_newer else FieldChange.Status.SUPERSEDED
65	            _write_history(change, record, name, value, status)
66	            if is_newer:
67	                _apply_field(change, record, name, value)
68	                applied_fields.append(name)
69	
70	        if created:
71	            record.created_by = change.actor
72	        if applied_fields:
73	            record.updated_by = change.actor
74	        if created or applied_fields:
75	            record.save()
76	
77	        if change.source == "portal" and applied_fields:
78	            _upsert_outbox_row(change.object_type, record.pk, applied_fields, change.operation)
79	
80	        return Result(outcome="applied" if applied_fields else "no_change", applied_fields=applied_fields)

83	def _is_echo(change: Change) -> bool:
84	    return change.source in SF_SOURCES and change.actor == settings.SALESFORCE_INTEGRATION_USER
85	
86	
87	def _lock_record(change: Change) -> tuple[Any, bool]:
88	    """Return (record, created). The row is locked FOR UPDATE until the transaction ends."""
89	    model = MODELS[change.object_type]
90	    qs = model.objects.select_for_update()
91	    try:
92	        if change.source == "portal":
93	            return qs.get(pk=change.record_ref), False
94	        # An echo never creates a record: a create echo for an unknown id is an error, not a new row.
95	        if change.operation == "create" and not _is_echo(change):
96	            return qs.get_or_create(salesforce_id=change.record_ref)
97	        return qs.get(salesforce_id=change.record_ref), False
98	    except model.DoesNotExist as exc:
99	        raise UnknownRecord(f"{change.object_type} {change.record_ref!r} does not exist") from exc
100

102	def _changed_fields(change: Change, record: Any) -> dict[str, Any]:
103	    if change.source == "portal":

105	    # Compare with the last value Salesforce sent for the field, not the current column (design §7).
106	    # A field with no sf_value yet counts as changed.
107	    changed = {}
108	    for name, value in change.fields.items():
109	        meta = record.field_meta.get(name, {})
110	        if "sf_value" not in meta or meta["sf_value"] != value:
111	            changed[name] = value
112	    return changed
