     8	# Atomic Lua token bucket:
     9	# Evaluates rate, refilling tokens continuously, and checking pause.
    10	# Returns:
    11	#   1: token acquired
    12	#  -1: paused (e.g. 429 slow down)
    13	#  >1: wait milliseconds until token available
    14	LUA_ACQUIRE = """
    15	local rate_key = KEYS[1]
    16	local tokens_key = KEYS[2]
    17	local refill_key = KEYS[3]
    18	local paused_key = KEYS[4]
    19	
    20	local default_rate = tonumber(ARGV[1])
    21	local now_ms = tonumber(ARGV[2])
    22	local cost = tonumber(ARGV[3])
    23	
    24	-- 1. Check if throttled/paused
    25	local paused_until = tonumber(redis.call('get', paused_key) or '0')
    26	if now_ms < paused_until then
    27	    return -1
    28	end
    29	
    30	-- 2. Current rate & capacity
    31	local rate = tonumber(redis.call('get', rate_key) or default_rate)
    32	local capacity = rate
    33	
    34	-- 3. Token refill based on elapsed time
    35	local tokens = tonumber(redis.call('get', tokens_key) or capacity)
    36	local last_refill = tonumber(redis.call('get', refill_key) or now_ms)
    37	local elapsed_ms = math.max(0, now_ms - last_refill)
    38	local refill = (elapsed_ms / 1000.0) * rate
    39	tokens = math.min(capacity, tokens + refill)
    40	redis.call('set', refill_key, now_ms)
    41	
    42	-- 4. Check availability
    43	if tokens >= cost then
    44	    tokens = tokens - cost
    45	    redis.call('set', tokens_key, tokens)
    46	    return 1
    47	else
    48	    redis.call('set', tokens_key, tokens)
    49	    local needed = cost - tokens
    50	    local wait_ms = math.ceil((needed / rate) * 1000)
    51	    return math.max(1, wait_ms)
    52	end
    53	"""

    92	    def acquire(self, timeout: float = 5.0, cost: int = 1) -> bool:
    93	        """Acquire `cost` tokens, blocking until available or `timeout` expires.
    94	
    95	        Returns True on success, False if timeout exceeded.
    96	        """
    97	        start = time.monotonic()
    98	        while True:
    99	            now_ms = int(self.clock() * 1000)
   100	            res = self._script(
   101	                keys=[self.rate_key, self.tokens_key, self.refill_key, self.paused_key],
   102	                args=[self.default_rate, now_ms, cost],
   103	            )
   104	            if res == 1:
   105	                return True
   106	
   107	            elapsed = time.monotonic() - start
   108	            remaining = timeout - elapsed
   109	            if remaining <= 0:
   110	                return False

   112	            if res == -1:  # paused

   114	            else:  # wait_ms
