2	
3	Pure Postgres: no network, no Salesforce client, no Redis.
4	"""
5	
6	from dataclasses import dataclass, field
7	from datetime import datetime
8	from typing import Any
9	
10	from django.conf import settings
11	from django.db import IntegrityError, models, transaction
12	from django.utils import timezone
13	
14	from sync.models import ChangeEvent, Contact, Customer, FieldChange, Outbox
15	
16	# "catch-up" is treated exactly like "salesforce" everywhere (AC-2.1.1).
17	SF_SOURCES = frozenset({"salesforce", "catch-up"})
18	MODELS = {"Customer": Customer, "Contact": Contact}
19	
20	
21	class UnknownRecord(LookupError):
22	    """A portal change or a Salesforce update names a record that does not exist. Raised rather
23	    than silently created (DEFAULTED, AC-2.1.2c); the transaction rolls back, so the ChangeEvent
24	    is not kept and a retry is not misreported as a duplicate."""
25	
26	
27	@dataclass(frozen=True)
28	class Change:
29	    source: str  # "salesforce" | "catch-up" | "portal"
30	    event_id: str
31	    object_type: str  # "Customer" | "Contact"
32	    record_ref: str | int  # salesforce_id (str) for salesforce/catch-up, pk (int) for portal
33	    occurred_at: datetime  # timezone-aware
34	    actor: str
35	    fields: dict[str, Any]
36	    operation: str  # "create" | "update"
37	
38	
39	@dataclass(frozen=True)
40	class Result:
41	    outcome: str  # "applied" | "no_change" | "duplicate" | "echo"
42	    applied_fields: list[str] = field(default_factory=list)
43	
44	
45	def apply(change: Change) -> Result:
46	    with transaction.atomic():
47	        try:
48	            # Savepoint, so a duplicate's IntegrityError doesn't poison the outer transaction.
49	            with transaction.atomic():
50	                ChangeEvent.objects.create(event_id=change.event_id, source=change.source)
51	        except IntegrityError:
52	            return Result(outcome="duplicate")
53	
54	        record, created = _lock_record(change)
55	
56	        if _is_echo(change):
57	            for name, value in change.fields.items():
58	                _write_history(change, record, name, value, FieldChange.Status.ECHO)
59	            return Result(outcome="echo")
60	
61	        applied_fields = []
62	        for name, value in _changed_fields(change, record).items():
63	            is_newer = _is_newer(change, record, name)
64	            status = FieldChange.Status.APPLIED if is_newer else FieldChange.Status.SUPERSEDED
65	            _write_history(change, record, name, value, status)
66	            if is_newer:
67	                _apply_field(change, record, name, value)
68	                applied_fields.append(name)
69	
70	        if created:
71	            record.created_by = change.actor
72	        if applied_fields:
73	            record.updated_by = change.actor
74	        if created or applied_fields:
75	            record.save()
76	            if change.object_type == "Customer" and record.salesforce_id:
77	                # When customer arrives, link waiting contacts (build-plan 3.1.4)
78	                Contact.objects.filter(
79	                    customer_salesforce_id=record.salesforce_id,
80	                    customer__isnull=True,
81	                ).update(customer=record)
82	
83	        if change.source == "portal" and applied_fields:
84	            _upsert_outbox_row(change.object_type, record.pk, applied_fields, change.operation)
85	
86	            def _notify_outbound():
87	                try:
88	                    from sync.tasks import send_outbound
89	                    send_outbound.send()
90	                except Exception:
91	                    pass
92	
93	            transaction.on_commit(_notify_outbound)
94	
95	        return Result(outcome="applied" if applied_fields else "no_change", applied_fields=applied_fields)
96	
97	
98	def _is_echo(change: Change) -> bool:
99	    return change.source in SF_SOURCES and change.actor == settings.SALESFORCE_INTEGRATION_USER
100	
101	
102	def _lock_record(change: Change) -> tuple[Any, bool]:
103	    """Return (record, created). The row is locked FOR UPDATE until the transaction ends."""
104	    model = MODELS[change.object_type]
105	    qs = model.objects.select_for_update()
106	    try:
107	        if change.source == "portal":
108	            return qs.get(pk=change.record_ref), False
109	        # An echo never creates a record: a create echo for an unknown id is an error, not a new row.
110	        if change.operation == "create" and not _is_echo(change):
111	            return qs.get_or_create(salesforce_id=change.record_ref)
112	        return qs.get(salesforce_id=change.record_ref), False
113	    except model.DoesNotExist as exc:
114	        raise UnknownRecord(f"{change.object_type} {change.record_ref!r} does not exist") from exc
115	
116	
117	def _changed_fields(change: Change, record: Any) -> dict[str, Any]:
118	    if change.source == "portal":
119	        return change.fields
120	    # Compare with the last value Salesforce sent for the field, not the current column (design §7).
121	    # A field with no sf_value yet counts as changed.
122	    changed = {}
123	    for name, value in change.fields.items():
124	        meta = record.field_meta.get(name, {})
125	        if "sf_value" not in meta or meta["sf_value"] != value:
126	            changed[name] = value
127	    return changed
128	
129	
130	def _is_newer(change: Change, record: Any, name: str) -> bool:
131	    stored = record.field_meta.get(name, {}).get("at")
132	    if stored is None:
133	        return True
134	    stored_at = datetime.fromisoformat(stored)  # stored as ISO string in JSON
135	    if change.occurred_at == stored_at:
136	        return change.source in SF_SOURCES  # exact tie: Salesforce wins (design §7)
137	    return change.occurred_at > stored_at
138	
139	
140	def _write_history(change: Change, record: Any, name: str, value: Any, status: str) -> None:
141	    if value is None:
142	        val_str = None
143	    elif isinstance(value, Customer):
144	        val_str = value.salesforce_id or str(value.pk)
145	    else:
146	        val_str = str(value)
147	
148	    FieldChange.objects.create(
149	        object_type=change.object_type,
150	        record_id=record.pk,
151	        field=name,
152	        value=val_str,
153	        actor=change.actor,
154	        source=change.source,
155	        occurred_at=change.occurred_at,
156	        status=status,
157	        event_id=change.event_id,
158	    )
159	
160	
161	def _apply_field(change: Change, record: Any, name: str, value: Any) -> None:
162	    old = record.field_meta.get(name, {})
163	    meta = {"at": change.occurred_at.isoformat(), "source": change.source, "by": change.actor}
164	    if change.source in SF_SOURCES:
165	        meta["sf_value"] = value
166	    elif "sf_value" in old:  # portal edit: keep what Salesforce last sent
167	        meta["sf_value"] = old["sf_value"]
168	    record.field_meta[name] = meta
169	
170	    if change.object_type == "Contact" and name == "customer":
171	        if change.source in SF_SOURCES:
172	            sf_cust_id = str(value) if value is not None else ""
173	            record.customer_salesforce_id = sf_cust_id
174	            if sf_cust_id:
175	                record.customer = Customer.objects.filter(salesforce_id=sf_cust_id).first()
176	            else:
177	                record.customer = None
178	            meta["sf_value"] = sf_cust_id
179	        else:
180	            if isinstance(value, Customer):
181	                record.customer = value
182	                record.customer_salesforce_id = value.salesforce_id or ""
183	            elif value:
184	                cust = Customer.objects.filter(pk=value).first()
185	                record.customer = cust
186	                record.customer_salesforce_id = cust.salesforce_id if cust else ""
187	            else:
188	                record.customer = None
189	                record.customer_salesforce_id = ""
190	    else:
191	        field_obj = record._meta.get_field(name)
192	        if value is None and getattr(field_obj, "null", False) is False:
193	            if hasattr(field_obj, "get_internal_type") and field_obj.get_internal_type() in (
194	                "CharField",
195	                "TextField",
196	                "EmailField",
197	            ):
198	                value = ""
199	        setattr(record, name, value)

205	        record_id=record_id,

210	        row.fields = sorted(set(row.fields) | set(applied_fields))
