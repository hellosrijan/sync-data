6	# Django and DRF conventions
7	
8	Applies to all code under `src/`. The how-to, with a worked example, is in the `drf-endpoint`

11	## Layout and URLs (user decision)
12	
13	- **Every app has `urls.py` (with `app_name`) and `serializers.py`.** Serializers don't live in
14	  `views.py`.
15	- **`config/urls.py` is the root URLconf and the only place that defines `api/v1/`.** It
16	  includes each app's `urls.py`. An app's `urls.py` never repeats `api/v1/`.
17	- Resource names are **singular**: `api/v1/customer/`, `api/v1/contact/`,
18	  `api/v1/customer/{id}/history/`. The contract is `docs/openapi.yml`.
19	- `/health/` is `api/v1/health/`.

21	## Views: DRF generic views only
22	
23	- Every endpoint is a class from `rest_framework.generics`: `ListAPIView`, `CreateAPIView`,
24	  `RetrieveAPIView`, `ListCreateAPIView`, `RetrieveUpdateAPIView`, and so on.
25	  `src/utils/views.py` (`HealthView`, a `RetrieveAPIView`) is the existing example.
26	- No function views (`@api_view`) and no bare `APIView`. If no generic class fits, subclass
27	  `generics.GenericAPIView` with the mixins you need, and write why in the task ledger.
28	- Views stay thin: queryset, serializer, permissions, pagination. Business rules live in
29	  services, the most important being `apply()`.
30	- Restrict methods with `http_method_names`. Updates are **PATCH only**, so a save sends just
31	  the changed fields, and only those reach history and the outbox.

34	
35	Customer and Contact are synced with Salesforce. **Never** write them with

39	history and stops the change from reaching Salesforce.

41	## Serializers: the trust boundary

43	- Explicit `fields`. Never `"__all__"`. Mark server-owned fields `read_only_fields`: IDs,

46	- Fields outside the brief's scope are ignored. For customers: name, status, owner. For

    def get_syncState(self, obj) -> str:
        # Check annotated or prefetched outbox state first to avoid N+1
        if hasattr(obj, "_sync_state"):
            return obj._sync_state
        # Fallback query
        obj_type = "Customer" if isinstance(obj, Customer) else "Contact"
        row = (
            Outbox.objects.filter(object_type=obj_type, record_id=obj.pk)
            .order_by("-id")
            .first()
        )
        if not row:
            return "synced"
        if row.state == Outbox.State.PENDING:
            return "pending"
        if row.state == Outbox.State.NEEDS_REVIEW:
            return "needs_review"
        return "synced"

class WebhookView(generics.CreateAPIView):
    """Receive one Salesforce change event over HTTPS POST (design §5, openapi.yml)."""

    authentication_classes = [TokenAuthentication]
    permission_classes = [IsSalesforceIntegration]
    serializer_class = WebhookEventSerializer

    def create(self, request, *args, **kwargs):
        serializer = self.get_serializer(data=request.data)
        serializer.is_valid(raise_exception=True)
        data = serializer.validated_data

        event_id = data["eventId"]
        record_id = data["recordId"]
        object_type = data["objectType"]
        occurred_at = data["occurredAt"]
        changed_by = data["changedBy"]
        raw_fields = data["changedFields"]

        allowed = ALLOWED_WEBHOOK_FIELDS[object_type]
        fields = {}
        for k, v in raw_fields.items():
            mapped_k = "full_name" if k == "fullName" else k
            if mapped_k in allowed:
                fields[mapped_k] = v

        model = Customer if object_type == "Customer" else Contact
        operation = "update" if model.objects.filter(salesforce_id=record_id).exists() else "create"

        change = Change(
            source="salesforce",
            event_id=event_id,
            object_type=object_type,
            record_ref=record_id,
            occurred_at=occurred_at,
            actor=changed_by,
            fields=fields,
            operation=operation,
        )
        result = apply(change)

        outcome = result.outcome
        if outcome == "no_change":
            if FieldChange.objects.filter(event_id=event_id, status=FieldChange.Status.SUPERSEDED).exists():
                outcome = "superseded"

        response_data = {
            "eventId": event_id,
            "result": outcome,
            "appliedFields": result.applied_fields,
        }
        result_serializer = WebhookResultSerializer(data=response_data)
        result_serializer.is_valid(raise_exception=True)
        return Response(result_serializer.data, status=status.HTTP_200_OK)


def _annotate_sync_state(queryset, object_type: str):
    """Annotate records with their syncState from Outbox to avoid N+1 queries."""
    pending_ids = set(
        Outbox.objects.filter(object_type=object_type, state=Outbox.State.PENDING).values_list("record_id", flat=True)
    )
    review_ids = set(
        Outbox.objects.filter(object_type=object_type, state=Outbox.State.NEEDS_REVIEW).values_list("record_id", flat=True)
    )

    records = list(queryset)
    for r in records:
        if r.pk in pending_ids:
            r._sync_state = "pending"
        elif r.pk in review_ids:
            r._sync_state = "needs_review"
        else:
            r._sync_state = "synced"
    return records

            operation="create",
        )

class UnknownRecord(LookupError):
    """A portal change or a Salesforce update names a record that does not exist. Raised rather
    than silently created (DEFAULTED, AC-2.1.2c); the transaction rolls back, so the ChangeEvent
    is not kept and a retry is not misreported as a duplicate."""


@dataclass(frozen=True)
class Change:
    source: str  # "salesforce" | "catch-up" | "portal"
    event_id: str
    object_type: str  # "Customer" | "Contact"
    record_ref: str | int  # salesforce_id (str) for salesforce/catch-up, pk (int) for portal
    occurred_at: datetime  # timezone-aware
    actor: str
    fields: dict[str, Any]
    operation: str  # "create" | "update"


@dataclass(frozen=True)
class Result:
    outcome: str  # "applied" | "no_change" | "duplicate" | "echo"
    applied_fields: list[str] = field(default_factory=list)


def apply(change: Change) -> Result:
    with transaction.atomic():
        try:
            # Savepoint, so a duplicate's IntegrityError doesn't poison the outer transaction.
            with transaction.atomic():
                ChangeEvent.objects.create(event_id=change.event_id, source=change.source)
        except IntegrityError:
            return Result(outcome="duplicate")

        record, created = _lock_record(change)

        if _is_echo(change):
            for name, value in change.fields.items():
                _write_history(change, record, name, value, FieldChange.Status.ECHO)
                if _is_newer(change, record, name):  # don't let a late echo overwrite a newer real value
                    meta = record.field_meta.get(name, {})
                    meta["sf_value"] = value
                    record.field_meta[name] = meta
            record.save(update_fields=["field_meta"])
            return Result(outcome="echo")

        applied_fields = []
        for name, value in _changed_fields(change, record).items():
            is_newer = _is_newer(change, record, name)
            status = FieldChange.Status.APPLIED if is_newer else FieldChange.Status.SUPERSEDED
            _write_history(change, record, name, value, status)
            if is_newer:
                _apply_field(change, record, name, value)
                applied_fields.append(name)

        if created or (change.source == "portal" and change.operation == "create" and not record.created_by):
            record.created_by = change.actor
        if applied_fields:
            record.updated_by = change.actor
        if created or applied_fields:

                record.customer_salesforce_id = ""

                    customer__isnull=True,

            def _notify_outbound():
                try:
                    from sync.tasks import send_outbound
                    send_outbound.send()
                except Exception:
                    pass

            transaction.on_commit(_notify_outbound)

        return Result(outcome="applied" if applied_fields else "no_change", applied_fields=applied_fields)


def _is_echo(change: Change) -> bool:
    return change.source in SF_SOURCES and change.actor == settings.SALESFORCE_INTEGRATION_USER


def _lock_record(change: Change) -> tuple[Any, bool]:
    """Return (record, created). The row is locked FOR UPDATE until the transaction ends."""
    model = MODELS[change.object_type]
    qs = model.objects.select_for_update()
    try:
        if change.source == "portal":
            return qs.get(pk=change.record_ref), False
        # An echo never creates a record: a create echo for an unknown id is an error, not a new row.
        if change.operation == "create" and not _is_echo(change):
            return qs.get_or_create(salesforce_id=change.record_ref)
        return qs.get(salesforce_id=change.record_ref), False
    except model.DoesNotExist as exc:
        raise UnknownRecord(f"{change.object_type} {change.record_ref!r} does not exist") from exc


def _changed_fields(change: Change, record: Any) -> dict[str, Any]:
    if change.source == "portal":
        return change.fields
    # Compare with the last value Salesforce sent for the field, not the current column (design §7).
    # A field with no sf_value yet counts as changed.
    changed = {}
    for name, value in change.fields.items():
        meta = record.field_meta.get(name, {})
        if "sf_value" not in meta or meta["sf_value"] != value:
            changed[name] = value
    return changed


def _is_newer(change: Change, record: Any, name: str) -> bool:
    stored = record.field_meta.get(name, {}).get("at")
    if stored is None:
        return True
    stored_at = datetime.fromisoformat(stored)  # stored as ISO string in JSON
    if change.occurred_at == stored_at:
        return change.source in SF_SOURCES  # exact tie: Salesforce wins (design §7)
    return change.occurred_at > stored_at

    if value is None:

    elif isinstance(value, Customer):
